cmake_minimum_required(VERSION 3.20)
project(boss_app_capture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(CAPTURE REQUIRED IMPORTED_TARGET x11 xcomposite xres xdamage xrender libsystemd)
    add_executable(boss-app-capture x11_capture.cpp)
    target_link_libraries(boss-app-capture PRIVATE PkgConfig::CAPTURE)
    target_compile_options(boss-app-capture PRIVATE -Wall -Wextra -Werror)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(boss-app-capture-test x11_capture_test.cpp)
        target_link_libraries(boss-app-capture-test PRIVATE PkgConfig::CAPTURE)
        target_compile_options(boss-app-capture-test PRIVATE -Wall -Wextra -Werror)
        add_test(NAME exact-x11-window COMMAND boss-app-capture-test)
    endif()
elseif(WIN32)
    add_executable(boss-app-capture windows_capture.cpp)
    set_property(TARGET boss-app-capture PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_definitions(boss-app-capture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_compile_options(boss-app-capture PRIVATE /W4 /WX /permissive-)
    target_link_libraries(boss-app-capture PRIVATE windowsapp d3d11 dxgi d3dcompiler dwmapi wtsapi32 user32 advapi32)
    include(CTest)
    if(BUILD_TESTING)
        add_executable(boss-app-capture-test windows_capture_test.cpp)
        set_property(TARGET boss-app-capture-test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_compile_definitions(boss-app-capture-test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
        target_compile_options(boss-app-capture-test PRIVATE /W4 /WX /permissive-)
        target_link_libraries(boss-app-capture-test PRIVATE d3d11 dxgi windowsapp user32)
        add_test(NAME exact-windows-window COMMAND boss-app-capture-test $<TARGET_FILE:boss-app-capture>)
        set_tests_properties(exact-windows-window PROPERTIES SKIP_RETURN_CODE 77)
        add_test(NAME reject-invalid-parent COMMAND boss-app-capture --probe 0)
        add_test(NAME reject-unknown-mode COMMAND boss-app-capture --invalid-mode 0)
        set_tests_properties(reject-invalid-parent reject-unknown-mode PROPERTIES WILL_FAIL TRUE)
    endif()
else()
    message(FATAL_ERROR "Native helper builds support Windows and Linux X11")
endif()
install(TARGETS boss-app-capture RUNTIME DESTINATION bin)
